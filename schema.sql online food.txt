SELECT * FROM customers;
SELECT * FROM vehicles;
SELECT * FROM service_requests;

SELECT
    c.id,
    c.name,
    COUNT(v.id) AS vehicle_count
FROM customers c
JOIN vehicles v ON c.id = v.customer_id
GROUP BY c.id, c.name
HAVING COUNT(v.id) > 1;

SELECT
    SUM(service_cost) AS total_service_revenue
FROM service_requests;

SELECT
    vehicle_id,
    COUNT(*) AS total_services
FROM service_requests
GROUP BY vehicle_id
ORDER BY total_services DESC;

SELECT
    strftime('%Y-%m', service_date) AS month,
    COUNT(*) AS total_services,
    SUM(service_cost) AS total_revenue
FROM service_requests
GROUP BY strftime('%Y-%m', service_date)
ORDER BY month;

SELECT *
FROM service_requests
WHERE status = 'Pending';

SELECT
    c.id,
    c.name,
    SUM(s.service_cost) AS total_spent,
    RANK() OVER (
        ORDER BY SUM(s.service_cost) DESC
    ) AS customer_rank
FROM customers c
JOIN vehicles v
    ON c.id = v.customer_id
JOIN service_requests s
    ON v.id = s.vehicle_id
GROUP BY c.id, c.name;
